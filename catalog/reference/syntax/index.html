Синтаксис языков программирования: сравнение

Как одна и та же конструкция выглядит в разных языках.

КонструкцияPythonC#JavaScriptSwift
Переменная
name = "Анна"
string name = "Анна";
let name = "Анна";
let name = "Анна"
Вывод
print("Привет")
Console.WriteLine("Привет");
console.log("Привет");
print("Привет")
Список
nums = [1, 2, 3]
var nums = new List<int> {1, 2, 3};
const nums = [1, 2, 3];
let nums = [1, 2, 3]
Условие
if x > 0:
    print('+')
else:
    print('-')
if (x > 0) Console.WriteLine("+");
else Console.WriteLine("-");
if (x > 0) console.log('+');
else console.log('-');
if x > 0 { print("+") } else { print("-") }
Цикл
for i in range(3):
    print(i)
for (int i = 0; i < 3; i++)
    Console.WriteLine(i);
for (let i = 0; i < 3; i++) {
  console.log(i);
}
for i in 0..<3 {
    print(i)
}
Функция
def add(a, b):
    return a + b
int Add(int a, int b) => a + b;
const add = (a, b) => a + b;
func add(_ a: Int, _ b: Int) -> Int {
    a + b
}
